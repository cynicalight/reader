package reader

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net"
	"net/http"
	"net/http/httptest"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"
)

func TestImprovementLive(t *testing.T) {
	if os.Getenv("READER_IMPROVEMENT_LIVE") != "1" {
		t.Skip("explicit live model opt-in required")
	}
	output := os.Getenv("READER_IMPROVEMENT_OUTPUT")
	if _, err := os.Stat(output); !os.IsNotExist(err) {
		t.Fatal("must use a new output file")
	}
	origin, err := os.ReadFile("improvement_context.txt")
	if err != nil {
		t.Fatal(err)
	}
	s := testServer(t)
	cfg := AIConfig{Primary: "codex", Models: map[string]string{"codex": "gpt-6.1-sol"}, Capabilities: map[string]Capability{}}
	cfg.Capabilities["codex"] = Capability{Text: true, Fingerprint: configPrint(cfg, "codex")}
	if err := s.writeAIConfig(cfg); err != nil {
		t.Fatal(err)
	}
	h := httptest.NewUnstartedServer(s.Handler())
	h.Listener, err = net.Listen("tcp4", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	h.Start()
	defer h.Close()
	defer func() { improvementAdapter = nil; improvementMode = "" }()
	client := &http.Client{Timeout: 4 * time.Minute}
	fetch := func(method, path string, body []byte) ([]byte, int) {
		r, err := http.NewRequest(method, h.URL+path, bytes.NewReader(body))
		if err != nil {
			t.Fatal(err)
		}
		r.Header.Set("Authorization", "Bearer test-secret")
		r.Header.Set("Content-Type", "application/json")
		res, err := client.Do(r)
		if err != nil {
			t.Fatal(err)
		}
		defer res.Body.Close()
		b, err := io.ReadAll(res.Body)
		if err != nil {
			t.Fatal(err)
		}
		return b, res.StatusCode
	}
	for block := 1; block <= 2; block++ {
		work := filepath.Join(s.Store.Root, "ai-work", fmt.Sprintf("session-%d", block))
		if err := os.MkdirAll(work, 0700); err != nil {
			t.Fatal(err)
		}
		session := &protoSession{server: s, work: work}
		defer session.close()
		marker := fmt.Sprintf("本次配对实验标识：%s/block-%d\n", time.Now().UTC().Format(time.RFC3339Nano), block)
		sourceA := marker + "独立事实 A：甲队预算为730元。\n" + string(origin)
		sourceB := marker + "独立事实 B：乙队预算为910元；甲队预算在此前资料A中，本资料不重复。\n"
		qs := []string{"请用一句话概括文章对缓存与连续对话的结论。", "缓存读取 token 为什么仍计入输入总量？一句话回答。", "甲队预算是多少？只用阿拉伯数字和单位回答。", "如果历史被截短，缓存可能受到什么影响？一句话回答。", "如何区分缓存命中与费用节省？一句话回答。", "结合前面的讨论，列出两个保持请求前缀稳定的做法。最多两点。"}
		if block == 2 {
			qs = []string{"资料A中甲队预算是多少？只用阿拉伯数字和单位回答。", "为什么缓存读取 token 仍计入输入总量？一句话回答。", "再次核对甲队预算，使用阿拉伯数字和单位。", "新资料B中乙队预算是多少？只用阿拉伯数字和单位回答。", "当前没有摘录。根据我们之前的对话，甲队和乙队预算分别是多少？使用阿拉伯数字和单位。", "仍没有当前摘录。甲队和乙队预算相差多少？使用阿拉伯数字和单位，并说明两个预算。"}
		}
		modes := []string{"baseline", "dedup", "session"}
		if block == 2 {
			modes = []string{"session", "dedup", "baseline"}
		}
		for _, mode := range modes {
			key := fmt.Sprintf("%s-%d", mode, block)
			if _, err := s.Store.DB.Exec("INSERT INTO documents(id,type,title,author,size,created_at,last_opened_at) VALUES(?,'pdf','合成文章：缓存与连续对话','',1,?,?)", key, now(), now()); err != nil {
				t.Fatal(err)
			}
		}
		previous := map[string]string{}
		for round, q := range qs {
			current := sourceA
			if block == 2 && round == 3 {
				current = sourceB
			}
			if block == 2 && round >= 4 {
				current = ""
			}
			for _, mode := range modes {
				key := fmt.Sprintf("%s-%d", mode, block)
				improvementMode = mode
				improvementAdapter = nil
				if mode == "session" {
					improvementAdapter = session
				}
				improvementThread = ""
				improvementSentPrompt = ""
				restarted := false
				if block == 2 && round == 5 && mode == "session" {
					session.close()
					session.restartReason = "explicit-process-restart"
					restarted = true
				}
				body, _ := json.Marshal(map[string]any{"provider": "codex", "prompt": q, "context": current})
				started := time.Now()
				b, status := fetch("POST", "/api/documents/"+key+"/chat", body)
				msg, _ := fetch("GET", "/api/documents/"+key+"/messages", nil)
				usage, _ := fetch("GET", "/api/documents/"+key+"/chat-usage", nil)
				var messages []Message
				_ = json.Unmarshal(msg, &messages)
				var report UsageReport
				_ = json.Unmarshal(usage, &report)
				answer := ""
				if len(messages) > 0 {
					answer = messages[len(messages)-1].Content
				}
				expected := []string{}
				if round == 2 || block == 2 && round == 0 {
					expected = []string{"730"}
				}
				if block == 2 && round == 3 {
					expected = []string{"910"}
				}
				if block == 2 && round == 4 {
					expected = []string{"730", "910"}
				}
				if block == 2 && round == 5 {
					expected = []string{"730", "910", "180"}
				}
				semantic := true
				for _, fragment := range expected {
					semantic = semantic && strings.Contains(answer, fragment)
				}
				var raw json.RawMessage
				resets := 0
				if mode == "session" {
					raw = session.rawUsage
					resets = session.resets
				}
				row := map[string]any{"block": block, "round": round + 1, "condition": mode, "question": q, "context": current, "answer": answer, "expectedFragments": expected, "semanticOK": semantic, "durationMs": time.Since(started).Milliseconds(), "messages": messages, "usage": report, "sse": string(b), "httpStatus": status, "builtPrompt": improvementBuiltPrompt, "sentPrompt": improvementSentPrompt, "threadId": improvementThread, "cwd": improvementCWD, "providerUsage": raw, "processStarts": resets, "explicitRestart": restarted, "previousBuiltAnswerPresent": round == 0 || strings.Contains(improvementBuiltPrompt, previous[key])}
				data, err := json.Marshal(row)
				if err != nil {
					t.Fatal(err)
				}
				f, err := os.OpenFile(output, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0600)
				if err != nil {
					t.Fatal(err)
				}
				_, err = f.Write(append(data, '\n'))
				_ = f.Close()
				if err != nil {
					t.Fatal(err)
				}
				if status != 200 || !strings.Contains(string(b), "event: done") || strings.Contains(string(b), "event: error") || strings.Contains(string(b), "event: fallback") || len(messages) != 2*(round+1) || len(report.Calls) != round+1 {
					t.Fatalf("failed %s round %d, evidence saved", key, round+1)
				}
				if !semantic {
					t.Errorf("semantic mismatch %s round %d: %q", key, round+1, answer)
				}
				if !strings.Contains(improvementBuiltPrompt, q) || (round > 0 && !strings.Contains(improvementBuiltPrompt, previous[key])) {
					t.Errorf("missing question or actual previous answer in Reader prompt %s round %d", key, round+1)
				}
				c := report.Calls[len(report.Calls)-1]
				if c.Status != "complete" || len(c.Models) != 1 || c.Models[0].Model != "gpt-6.1-sol" || c.Models[0].Tokens == nil {
					t.Fatal("invalid persisted usage")
				}
				v := c.Models[0].Tokens
				t.Logf("block=%d mode=%s round=%d input=%d output=%d cache=%d semantic=%v", block, mode, round+1, v.InputTokens, v.OutputTokens, *v.CachedInputTokens, semantic)
				previous[key] = answer
			}
		}
		session.close()
	}
}
